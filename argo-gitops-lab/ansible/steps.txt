master:

sudo apt-get update
sudo apt-get install -y apt-transport-https ca-certificates curl gpg

# Create the directory securely
sudo mkdir -p -m 755 /etc/apt/keyrings

# Download and de-armor the GPG key
if [ ! -f /etc/apt/keyrings/kubernetes-apt-keyring.gpg ]; then
  echo "Keyring file missing. Safe to download..."
  curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.30/deb/Release.key | \
  sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
else
  echo "Kubernetes GPG keyring already exists. Skipping download."
fi

echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.30/deb/ /' | sudo tee /etc/apt/sources.list.d/kubernetes.list

sudo apt-get update
sudo apt-get install -y kubeadm kubelet kubectl


# Update your packet list and install both packages
sudo apt-get update
sudo apt-get install -y containerd conntrack

# containerd config
# Create the directory for containerd's configuration file
sudo mkdir -p /etc/containerd
# Generate the default configuration text and save it to the file
sudo containerd config default | sudo tee /etc/containerd/config.toml
sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/g' /etc/containerd/config.toml

# 1. Restart containerd to load the systemd cgroup driver setting
sudo systemctl restart containerd

# 2. Verify containerd is running smoothly
sudo systemctl status containerd

# 1. Configure the system variable permanently
echo "net.ipv4.ip_forward = 1" | sudo tee /etc/sysctl.d/k8s.conf

# 2. Apply the change right now without rebooting
sudo sysctl --system

# enable kubelet
sudo systemctl enable --now kubelet

sudo kubeadm init --pod-network-cidr=192.168.0.0/16

TARGET_API_SERVER=$(hostname -I | awk '{print $1}')
KUBECONFIG_FILE="$HOME/.kube/config"

echo "Detected Master Node IP: $TARGET_API_SERVER"

# 1. Check if the kubeconfig exists AND can successfully reach the specific control plane
if [ -f "$KUBECONFIG_FILE" ] && kubectl --kubeconfig="$KUBECONFIG_FILE" cluster-info &>/dev/null && grep -q "$TARGET_API_SERVER" "$KUBECONFIG_FILE"; then
  echo "Control plane is already initialized and active at $TARGET_API_SERVER. Skipping init." [cite: 124, 125]
else
  echo "Control plane not initialized or unreachable. Preparing to initialize..." [cite: 124]
  
  # 2. Safety cleanup: If an old or broken cluster exists, reset it first
  if [ -f /etc/kubernetes/admin.conf ]; then
    echo "Detected old configuration files. Resetting kubeadm state..." [cite: 126]
    sudo kubeadm reset -f [cite: 126]
    rm -rf "$HOME/.kube"
  fi

  # 3. Initialize the master node (passing the dynamically discovered IP to the API server)
  sudo kubeadm init --pod-network-cidr=192.168.0.0/16 --apiserver-advertise-address="$TARGET_API_SERVER"
  
  # 4. Automatically configure kubeconfig for the local user
  mkdir -p "$HOME/.kube"
  sudo cp -i /etc/kubernetes/admin.conf "$KUBECONFIG_FILE"
  sudo chown "$(id -u):$(id -g)" "$KUBECONFIG_FILE"
  echo "Control plane successfully initialized!"
fi

TARGET_API_SERVER=$(hostname -I | awk '{print $1}')
KUBECONFIG_FILE="$HOME/.kube/config"

echo "Detected Master Node IP: $TARGET_API_SERVER"

# Install helm: Download and run the installer script
curl -fsSL -o get_helm.sh https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3
chmod 700 get_helm.sh
./get_helm.sh

# Add the official Tigera Calico chart URL
helm repo add projectcalico https://docs.tigera.io/calico/charts

# Sync your local index with the server
helm repo update

DEFAULT_INTERFACE=$(ip route show default | awk '{print $5}')

helm install calico projectcalico/tigera-operator \
  --version v3.28.0 \
  --namespace tigera-operator \
  --create-namespace \
  --set installation.calicoNetwork.nodeAddressAutodetectionV4.interface=$DEFAULT_INTERFACE

TARGET_HASH="28f1efcf019dd8432490d308f3c7d93219808a6e5ce568f8e267675745f01cf3"
LOCAL_CA="/etc/kubernetes/pki/ca.crt"

# Check if the file exists and its hash matches your target cluster
if [ -f "$LOCAL_CA" ] && [ "$(openssl x509 -pubkey -in "$LOCAL_CA" | openssl rsa -pubin -outform der 2>/dev/null | openssl dgst -sha256 -hex | sed 's/^.* //')" = "$TARGET_HASH" ]; then
  echo "Node is already actively verified with this specific cluster. Skipping join."

    echo "Old cluster configuration detected. Resetting node first..."
    sudo kubeadm reset -f
  fi
  
  sudo kubeadm join 10.0.1.203:6443 --token bg88d5.uj1i85u1ju36u4zo --discovery-token-ca-cert-hash sha256:$TARGET_HASH
